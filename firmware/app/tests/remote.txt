# REMOTE MODE (docs/BROWSER-FM1-PLAN.md; app.c fz_rm_*): the unit as the panel of the FM-1 in the browser, taken by a
# RemoteFM1 host. HELLO and SESSION; the buttons, keys and knobs out as RemoteFM1 sends them (buttons: notes on channel
# 15 by matrix id; keys: notes on channel 16 from 53; knobs: CC 16.. on channel 15), the host's lights, EVILGENIUS
# itself deaf to the panel meanwhile; the session's end and its keepalive.
#   firmware/build.py --host, then firmware/app/build/host/fizz_host firmware/app/tests/remote.txt OUT
expect fz_up 1
tap HOME
wait 300
expect fz_rm 0
# HELLO alone takes nothing
sysex F0 7D 52 46 01 F7
wait 20
expect fz_rm 0
# SESSION start, keepalive 3000 ms (01 B8 0B, 7-in-8)
sysex F0 7D 52 46 10 02 01 38 0B F7
wait 50
expect fz_rm 1
expect fz_rm_n 1
shot remote-mode.png
out
# the panel out: FX (matrix id 2), PLAY (12), OCT- (0), key 5 (note 58), KNOB 1 up (matrix encoder 2: CC 18) and PRESETS down (6: CC 22), a detent a message here
tap FX
tap PLAY
tap OCT-
tapkey 5
turn K1 3
turn PRESET -2
wait 100
expect_out USB 9E 02 7F
expect_out USB 8E 02 00
expect_out USB 9E 0C 7F
expect_out USB 9E 00 7F
expect_out USB 9F 3A 64
expect_out USB 8F 3A 00
expect_out USB BE 12 41
expect_out USB BE 16 3F
# EVILGENIUS itself heard none of it: PLAY did not start it
expect_no_out USB FA 00 00
out
# the host's lights: FX lit (column 6, row PA5) and PLAY's green (the flags)
sysex F0 7D 52 46 11 00 00 00 00 00 00 00 02 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 01 F7
wait 50
expect fz_rm_leds 16385
leds
# SESSION end: EVILGENIUS again, PLAY plays
sysex F0 7D 52 46 10 00 00 00 00 F7
wait 100
expect fz_rm 0
shot remote-after.png
tap PLAY
wait 300
expect_out USB FA 00 00
tap PLAY
wait 200
out
# a session nobody keeps alive (200 ms) ends by itself
sysex F0 7D 52 46 10 02 01 48 00 F7
wait 60
expect fz_rm 1
# (another host's HELLOs, a RemoteFM1 app looking for units, do not keep it alive)
sysex F0 7D 52 46 01 F7
wait 150
sysex F0 7D 52 46 01 F7
wait 150
sysex F0 7D 52 46 01 F7
wait 300
expect fz_rm 0
expect fz_rm_n 2
